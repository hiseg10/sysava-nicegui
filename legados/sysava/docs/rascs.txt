CREATE TABLE IF NOT EXISTS public.app_users (

    password TEXT NOT NULL,
    name TEXT NOT NULL,
    ra TEXT,
    role TEXT DEFAULT 'student'
);

-- 2. Estrutura Acadêmica (Escolas, Turmas, Disciplinas)
CREATE TABLE IF NOT EXISTS public.schools (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    gre TEXT
);

CREATE TABLE IF NOT EXISTS public.classes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    code TEXT UNIQUE NOT NULL,
    school_id BIGINT REFERENCES public.schools(id)
);

CREATE TABLE IF NOT EXISTS public.subjects (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT UNIQUE NOT NULL
);

-- Tabela de junção: Turmas <-> Disciplinas
CREATE TABLE IF NOT EXISTS public.class_subjects (
    class_id BIGINT REFERENCES public.classes(id),
    subject_id BIGINT REFERENCES public.subjects(id),
    PRIMARY KEY (class_id, subject_id)
);

-- Matrículas dos alunos
CREATE TABLE IF NOT EXISTS public.student_enrollments (
    user_username TEXT REFERENCES public.app_users(username),
    class_id BIGINT REFERENCES public.classes(id),
    PRIMARY KEY (user_username, class_id)
);

-- 3. Conteúdo de Ensino (Aulas, Fórum, Quiz)
CREATE TABLE IF NOT EXISTS public.lessons (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title TEXT NOT NULL,
    description TEXT,
    video_url TEXT,
    subject_id BIGINT REFERENCES public.subjects(id),
    -- Garante que não existam duas aulas com mesmo título na mesma disciplina
    CONSTRAINT lessons_subject_id_title_key UNIQUE (subject_id, title)
);

CREATE TABLE IF NOT EXISTS public.forum_posts (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_name TEXT NOT NULL,
    message TEXT NOT NULL,
    lesson_id BIGINT REFERENCES public.lessons(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.quizzes (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    lesson_id BIGINT REFERENCES public.lessons(id) ON DELETE CASCADE,
    title TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS public.quiz_questions (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    quiz_id BIGINT REFERENCES public.quizzes(id) ON DELETE CASCADE,
    question_text TEXT NOT NULL,
    options JSONB NOT NULL, -- Ex: ["Opção A", "Opção B", "Opção C"]

);

🏆 Melhor Escolha (Equilíbrio entre Qualidade e Velocidade)
Mistral 7B Instruct v0.3

Por quê? Este é um dos modelos de 7 bilhões de parâmetros mais respeitados e eficientes. A palavra "Instruct" significa que ele foi treinado especificamente para seguir instruções, que é exatamente o que nosso script faz ao enviar o prompt. Ele vai entender o template da aula e preenchê-lo de forma muito competente, oferecendo o melhor equilíbrio entre a qualidade da aula gerada e a velocidade de execução nos seus 16GB de RAM.
🚀 Alternativa Mais Rápida
Phi 4

Por quê? Se você testar o Mistral 7B e ainda achar que está lento, o Phi 4 é a sua melhor aposta. A família de modelos "Phi" da Microsoft é famosa por ser muito "inteligente" para o seu tamanho reduzido. Ele provavelmente será o mais rápido da lista e consumirá menos RAM, mas ainda assim entregará um resultado de boa qualidade para a geração das aulas.
🧠 Alternativa de Maior Qualidade (Pode ser mais lento)
Gemma 2 9B

Por quê? Sendo o modelo mais recente do Google nesta categoria, ele provavelmente terá a maior "inteligência" e capacidade de raciocínio entre as opções viáveis. No entanto, com 9 bilhões de parâmetros, ele vai exigir mais da sua RAM e GPU, resultando em um tempo de geração maior. Se você não se importar em esperar um pouco mais por uma aula potencialmente mais elaborada, vale a pena testá-lo.
Modelos a Evitar para o seu caso:
Qwen2.5 VL 7B: O "VL" significa "Vision-Language". Ele é feito para entender imagens e texto, o que é um desperdício de recursos para a sua tarefa, que é 100% texto.
Codestral 22B, Qwen2.5 Coder (14B e 32B): São modelos excelentes, mas especializados em gerar código de programação. Eles não são otimizados para criar textos criativos e estruturados como um plano de aula.
Mistral Nemo 12B, Gemma 2 27B: São modelos muito grandes e com certeza serão lentos demais ou nem carregarão com 16GB de RAM.
Resumo da recomendação:

Comece baixando o Mistral 7B Instruct v0.3. É a aposta mais segura.
Se precisar de mais velocidade, troque para o Phi 4.
Se quiser a máxima qualidade e não se importar com a espera, experimente o Gemma 2 9B.
Lembre-se de sempre procurar pelas versões quantizadas (ex: Q4_K_M.gguf) no LM Studio. Elas são otimizadas para rodar com menos RAM.
